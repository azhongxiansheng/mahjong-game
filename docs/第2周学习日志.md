**任务1：创建卡牌属性脚本**

创建 `res://scripts/card_data.gd`：

```gdscript
extends Node

class_name CardData

# 卡牌类型枚举
enum Suit { WAN, TONG, TIAO, ZI }  # 万、筒、条、字

# 卡牌属性
var suit: Suit
var number: int
var id: int

func _init(p_suit: Suit, p_number: int) -> void:
	suit = p_suit
	number = p_number
	id = int(suit) * 10 + p_number

func get_name() -> String:
	var suit_name = ["万", "筒", "条", "字"][suit]
	return suit_name + str(number)
```

func can_pair_with(other: CardData) -> bool:

	elif suit > other.suit:

### 第4-5天（Day 4-5）：循环（for/while）（2-2.5小时）

#### 📚 学习内容

```gdscript
# for 循环遍历数组
for card in cards:
	print(card.get_name())

# for 循环指定次数
for i in range(5):  # i 从 0 到 4
	print("次数 %d" % i)

# for 循环指定范围
for i in range(1, 10):  # i 从 1 到 9
	print(i)

# while 循环
var count = 0
while count < 5:
	print("计数: " + str(count))
	count += 1

# break 和 continue
for i in range(10):
	if i == 3:
		continue  # 跳过 3
	if i == 7:
		break     # 在 7 处停止
	print(i)

# 嵌套循环
for suit in range(4):
	for number in range(1, 10):
		print("卡牌: %d-%d" % [suit, number])
```

#### 🎯 实践任务

**任务：实现卡牌组管理**

创建 `res://scripts/card_hand.gd`：

```gdscript
extends Node

class_name CardHand

var cards: Array[CardData] = []

# 添加卡牌
func add_card(card: CardData) -> void:
	cards.append(card)
	print("添加卡牌: " + card.get_name())

# 查找卡牌
func find_card(suit: int, number: int) -> CardData:
	for card in cards:
		if card.suit == suit and card.number == number:
			return card
	return null

# 移除卡牌
func remove_card(card: CardData) -> bool:
	for i in range(cards.size()):
		if cards[i] == card:
			cards.remove_at(i)
			return true
	return false

# 获取所有同花色的卡牌
func get_cards_by_suit(suit: int) -> Array[CardData]:
	var result: Array[CardData] = []
	for card in cards:
		if card.suit == suit:
			result.append(card)
	return result

# 排序卡牌
func sort_cards() -> void:
	cards.sort_custom(func(a: CardData, b: CardData) -> bool:
		if a.suit != b.suit:
			return a.suit < b.suit
		return a.number < b.number
	)

# 显示手牌
func print_hand() -> void:
	print("当前手牌:")
	for card in cards:
		print("  - " + card.get_name())
```
