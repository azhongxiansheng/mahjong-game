```gdscript
# 创建 godot/scripts/card_tile.gd

class_name CardTile
extends Control

# 属性
var card_data: CardData
var is_selected: bool = false
var is_highlighted: bool = false

# UI引用
@onready var card_label = $Label
@onready var suit_label = $SuitLabel

# 信号
signal card_pressed(card_data: CardData)
signal card_selected(card_data: CardData)

func _ready() -> void:
    gui_input.connect(_on_gui_input)
    mouse_entered.connect(_on_mouse_entered)
    mouse_exited.connect(_on_mouse_exited)
    custom_minimum_size = Vector2(80, 120)

func set_card(card: CardData) -> void:
    card_data = card
    update_display()

func update_display() -> void:
    if not card_data:
        return
    card_label.text = str(card_data.number)
    suit_label.text = _get_suit_name(card_data.suit)
    modulate = _get_suit_color(card_data.suit)

func _get_suit_name(suit: int) -> String:
    match suit:
        0: return "万"
        1: return "筒"
        2: return "条"
        3: return "字"
    return ""

func _get_suit_color(suit: int) -> Color:
    match suit:
        0: return Color.RED
        1: return Color.BLUE
        2: return Color.GREEN
        3: return Color.YELLOW
    return Color.WHITE

func highlight() -> void:
    is_highlighted = true
    self_modulate = Color.WHITE

func unhighlight() -> void:
    is_highlighted = false
    self_modulate = Color(0.7, 0.7, 0.7)

func select() -> void:
    is_selected = true
    scale = Vector2(1.1, 1.1)
    card_selected.emit(card_data)

func deselect() -> void:
    is_selected = false
    scale = Vector2(1.0, 1.0)

func _on_gui_input(event: InputEvent) -> void:
    if event is InputEventMouseButton and event.pressed:
        card_pressed.emit(card_data)

func _on_mouse_entered() -> void:
    highlight()

func _on_mouse_exited() -> void:
    if not is_selected:
        unhighlight()
```

# 创建 godot/scripts/hand_display.gd

class_name HandDisplay
extends GridContainer

# 属性
var hand: CardHand
var card_tiles: Array[CardTile] = []
var selected_tile: CardTile = null

# 引用
var card_tile_scene: PackedScene = preload("res://scenes/card_tile.tscn")

func _ready() -> void:
    columns = 13

func set_hand(h: CardHand) -> void:
    hand = h
    refresh_display()

func refresh_display() -> void:
    # 清空旧的卡牌显示
    for tile in card_tiles:
        tile.queue_free()
    card_tiles.clear()
    
    # 创建新的卡牌显示
    if not hand:
        return
    
    for card in hand.cards:
        add_card_display(card)

func add_card_display(card: CardData) -> void:
    var tile: CardTile = card_tile_scene.instantiate()
    tile.set_card(card)
    add_child(tile)
    
    tile.card_pressed.connect(_on_card_pressed.bind(tile))
    tile.card_selected.connect(_on_card_selected.bind(tile))
    card_tiles.append(tile)

func remove_card_display(card: CardData) -> void:
    for tile in card_tiles:
        if tile.card_data == card:
            tile.queue_free()
            card_tiles.erase(tile)
            break

func select_card(tile: CardTile) -> void:
    # 取消之前的选择
    if selected_tile and selected_tile != tile:
        selected_tile.deselect()
    
    # 选择新卡牌
    selected_tile = tile
    tile.select()

func get_selected_card() -> CardData:
    if selected_tile:
        return selected_tile.card_data
    return null

func _on_card_pressed(tile: CardTile) -> void:
    select_card(tile)

func _on_card_selected(tile: CardTile) -> void:
    # 可选: 触发出牌确认
    pass

#### 步骤3.2: 编写GameUI脚本
```gdscript
# 创建 godot/scripts/game_ui.gd

class_name GameUI
extends ScreenBase

# 组件引用
@onready var player_hand_display = $GameLayer/TableArea/PlayerHand
@onready var opponent_hand_display = $GameLayer/TableArea/OpponentHand
@onready var discard_pile = $GameLayer/TableArea/GameCenter/DiscardPile
@onready var game_info = $GameLayer/TableArea/GameCenter/GameInfo
@onready var player_stats = $InfoPanel/PlayerStats
@onready var game_log = $InfoPanel/GameLog

@onready var hu_button = $GameLayer/ActionPanel/HuButton
@onready var ting_button = $GameLayer/ActionPanel/TingButton
@onready var peng_button = $GameLayer/ActionPanel/PengButton
@onready var pass_button = $GameLayer/ActionPanel/PassButton
@onready var quit_button = $InfoPanel/QuitButton

# 游戏状态
var game_controller: GameController
var current_hand: CardHand

# 信号
signal card_played(card: CardData)
signal player_action(action: String)

func _ready() -> void:
    super()
    setup_ui()
    connect_signals()

func setup_ui() -> void:
    # 设置初始状态
    player_stats.text = "等待游戏开始..."
    game_log.text = ""

func connect_signals() -> void:
    # 连接按钮信号
    hu_button.pressed.connect(_on_hu_pressed)
    ting_button.pressed.connect(_on_ting_pressed)
    peng_button.pressed.connect(_on_peng_pressed)
    pass_button.pressed.connect(_on_pass_pressed)
    quit_button.pressed.connect(_on_quit_pressed)
    
    # 连接手牌信号
    player_hand_display.card_tiles.connect(_on_card_pressed)

func display_hand(hand: CardHand) -> void:
    current_hand = hand
    player_hand_display.set_hand(hand)
    update_player_stats()

func add_log_message(message: String) -> void:
    game_log.text += message + "\n"
    game_log.scroll_to_line(game_log.get_line_count() - 1)

func update_player_stats() -> void:
    player_stats.text = "手牌数: %d" % current_hand.get_card_count()

func _on_hu_pressed() -> void:
    if current_hand:
        var result = WinChecker.check_win(current_hand)
        if result.can_win:
            add_log_message("✓ 胡牌成功！")
            show_win_result(result)
        else:
            add_log_message("✗ 无法胡牌")

func _on_ting_pressed() -> void:
    if current_hand:
        var ting_result = TingChecker.check_ting(current_hand)
        if ting_result.can_ting:
            add_log_message("✓ 可以听牌，听 %d 种牌" % ting_result.ting_count)
        else:
            add_log_message("✗ 无法听牌")

func _on_peng_pressed() -> void:
    add_log_message("碰操作")

func _on_pass_pressed() -> void:
    add_log_message("跳过操作")

func _on_quit_pressed() -> void:
    get_tree().quit()

func show_win_result(result: WinResult) -> void:
    add_log_message("将: %s" % result.eye_card.get_card_name() if result.eye_card else "")
    for pattern in result.win_patterns:
        add_log_message("胡牌类型: %s" % str(pattern))
```

## 💡 提示和常见问题

### Q: 卡牌显示不出来？
**A**: 检查CardTile场景是否正确保存，脚本是否正确关联

### Q: 按钮没有响应？
**A**: 确保信号连接正确，检查_on_XXX函数名称是否匹配

### Q: 动画不流畅？
**A**: 使用Tween而不是直接修改position，调整duration参数
