### 第1-2天（Day 1-2）：函数进阶（2-2.5小时）

#### 📚 学习内容

**1. 默认参数值**

```gdscript
# 不使用默认参数
func create_card(suit: int, number: int, face_up: bool) -> CardData:
	var card = CardData.new(suit, number)
	return card

# 使用默认参数
func create_card(suit: int, number: int, face_up: bool = true) -> CardData:
	var card = CardData.new(suit, number)
	return card

# 调用时可以省略参数
var card1 = create_card(0, 5)  # face_up 默认为 true
var card2 = create_card(0, 5, false)  # 明确指定 face_up
```

**2. 可变参数**

```gdscript
# 接受任意数量的参数
func add_multiple_cards(cards: Array) -> void:
	for card in cards:
		hand.add_card(card)

# 使用
var cards = [card1, card2, card3]
add_multiple_cards(cards)

# 或者用 varargs（变长参数）
func print_cards(args: Array) -> void:
	for item in args:
		print(item)
```

**3. 函数返回多个值**

```gdscript
# 返回字典
func get_hand_info() -> Dictionary:
	return {
		"count": hand.get_card_count(),
		"suits": count_by_suit(),
		"ranks": count_by_rank()
	}

# 使用
var info = get_hand_info()
print("卡牌数: %d" % info["count"])
```

### 第3-4天（Day 3-4）：类的继承（2-2.5小时）

#### 📚 学习内容

**1. 基类和派生类**

```gdscript
# 基类
class_name BaseCard

var suit: int
var number: int

func _init(p_suit: int, p_number: int) -> void:
	suit = p_suit
	number = p_number

func get_value() -> int:
	return number

# 派生类（继承 BaseCard）
class_name SpecialCard extends BaseCard

var is_special: bool = true
var bonus: int = 0

func _init(p_suit: int, p_number: int, p_bonus: int = 0) -> void:
	super._init(p_suit, p_number)  # 调用父类构造函数
	bonus = p_bonus

func get_value() -> int:
	return number + bonus  # 覆盖父类方法
```

**2. super 关键字**

```gdscript
class_name ExtendedCard extends CardData

var special_effect: String = ""

func _init(p_suit: int, p_number: int) -> void:
	super._init(p_suit, p_number)  # 调用父类初始化

func get_card_name() -> String:
	var base_name = super.get_card_name()  # 调用父类方法
	if special_effect:
		return base_name + " [%s]" % special_effect
	return base_name
```

**3. is 和 as 操作符**

```gdscript
var card: CardData = CardData.new(0, 5)
var special_card: SpecialCard = SpecialCard.new(0, 5, 10)

# 类型检查
if card is CardData:
	print("是卡牌")

if special_card is SpecialCard:
	print("是特殊卡牌")

# 类型转换
if card is SpecialCard:
	var sc = card as SpecialCard
	print(sc.bonus)
```

### 第5天（Day 5）：卡牌组生成和牌池管理（2-2.5小时）

#### 📚 学习内容

**卡牌组（Deck）设计**

```gdscript
class_name CardDeck

var cards: Array = []
var discard_pile: Array = []  # 弃牌堆

func _init() -> void:
	"""初始化空牌池"""
	cards = []
	discard_pile = []

func add_card(card: CardData) -> void:
	"""添加卡牌到牌池"""
	cards.append(card)

func add_cards(new_cards: Array) -> void:
	"""添加多张卡牌"""
	for card in new_cards:
		cards.append(card)

func draw_card() -> CardData:
	"""从牌池抽一张卡"""
	if cards.size() == 0:
		print("牌池为空！")
		return null
	return cards.pop_front()

func draw_cards(count: int) -> Array:
	"""抽多张卡"""
	var result: Array = []
	for i in range(count):
		if cards.size() > 0:
			result.append(draw_card())
	return result

func shuffle() -> void:
	"""洗牌"""
	cards.shuffle()

def get_remaining_count() -> int:
	"""获取剩余卡牌数"""
	return cards.size()
```

extends CardDeck

class_name MahjongDeck

func _init() -> void:
	"""初始化标准麻将牌池（108张）"""
	super._init()
	generate_standard_deck()

func generate_standard_deck() -> void:
	"""生成标准麻将牌（万、筒、条各36张，共108张）"""
	# 万牌：1-9，每种 4 张 = 36张
	# 筒牌：1-9，每种 4 张 = 36张
	# 条牌：1-9，每种 4 张 = 36张
	# 共计 108 张
	for suit in range(3):
		for number in range(1, 10):
			for count in range(4):
				var card = CardData.new(suit, number)
				add_card(card)
	print("麻将牌池已生成：%d 张卡牌" % cards.size())

func draw_hand(count: int = 13) -> Array:
	"""抽初始手牌"""
	return draw_cards(count)

func get_deck_info() -> Dictionary:
	"""获取牌池信息"""
	return {
		"total": 108,
		"remaining": cards.size(),
		"discarded": discard_pile.size()
	}
