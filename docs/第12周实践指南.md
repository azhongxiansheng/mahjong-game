### 任务1: 实现弃牌堆显示系统 (1.5小时)

**目标**: 实现4x4网格的弃牌堆显示

#### 步骤1.1: 在GameUI中添加弃牌堆管理方法

在 `game_ui.gd` 中添加以下代码：

```gdscript
func update_discard_pile_display() -> void:
	"""更新弃牌堆显示"""
	if not discard_pile:
		return
	
	# 清空旧的显示
	for child in discard_pile.get_children():
		child.queue_free()
	
	# 显示最近的16张牌（4x4网格）
	var display_count = min(16, discard_cards.size())
	var start_index = max(0, discard_cards.size() - 16)
	
	for i in range(display_count):
		var card = discard_cards[start_index + i]
		var label = Label.new()
		label.text = card.get_card_name()
		label.add_theme_font_size_override("font_size", 12)
		label.custom_minimum_size = Vector2(50, 40)
		label.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
		label.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
		discard_pile.add_child(label)
```

#### 步骤3.1: 创建game_flow.gd文件

```gdscript
class_name GameFlow
extends Node

var game_controller: GameController
var game_ui: GameUI

var current_round: int = 0
var current_player_index: int = 0
var game_started: bool = false
var game_paused: bool = false

signal game_initialized
signal round_started(round_number: int)
signal turn_started(player_index: int)
signal turn_ended
signal game_ended(winner_index: int)

func _ready() -> void:
	print("========== GameFlow 初始化 ==========")
	game_ui = get_tree().root.get_node_or_null("Main/GameUI")
	if game_ui:
		print("✓ GameFlow已初始化")

func init_game(player_count: int = 4) -> void:
	"""初始化游戏"""
	print("\n========== 初始化游戏 ==========")
	
	current_round = 1
	current_player_index = 0
	game_started = true
	
	game_initialized.emit()
	await get_tree().create_timer(0.5).timeout
	start_round()

func start_round() -> void:
	"""开始新的一轮"""
	print("\n========== 第 %d 轮开始 ==========" % current_round)
	
	if game_ui:
		game_ui.add_log_message("✓ 第 %d 轮开始" % current_round)
	
	round_started.emit(current_round)
	
	await get_tree().create_timer(0.5).timeout
	start_turn()

# ... 其他方法 ...
```

	"""检查是否能胡牌"""
	if not WinChecker:
		print("⚠ WinChecker不可用")

	var result = WinChecker.check_win(hand)

## 💡 提示和常见问题

### Q: 弃牌堆显示不出来？
**A**: 检查discard_pile节点是否在场景中正确创建

### Q: 对手手牌为空？
**A**: 确保display_opponent_hand()传入了正确的卡牌数量

### Q: GameFlow无法找到GameUI？
**A**: 检查节点路径是否正确，使用get_node_or_null()进行安全查询

### Q: 胡牌检测不工作？
**A**: 确保WinChecker脚本存在且check_win()方法可用
