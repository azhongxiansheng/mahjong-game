【第 1 步】打开浏览器
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

在地址栏粘贴：

https://railway.app/dashboard

按 Enter


【第 2 步】登录（如果需要）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

用 GitHub 账户登录：azhongxiansheng


【第 3 步】找到项目
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

左侧看到红色的 mahjong-game (Failed)

点击这个项目


【第 4 步】进入 Settings
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

顶部导航栏最右边

点击【Settings】


【第 5 步】找到 Start Command
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

向下滚动页面

找到输入框：【Start Command】


【第 6 步】输入启动命令
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

点击 Start Command 输入框

清空里面的内容（Ctrl+A，Delete）

输入这个命令（完全复制）：

cd backend && go run main.go


【第 7 步】保存
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

页面上找绿色的【Save】或【Update】按钮

点击保存


【第 8 步】返回项目
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

点击左上角返回或点项目名称


【第 9 步】重新部署
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

找【Redeploy】或【Relaunch】按钮

点击它

等待 5-10 分钟...
